create table if not exists public.yofista_views ( id bigint generated by default as identity primary key, post_id uuid not null references public.yofista_posts(id) on delete cascade, viewer_id uuid null references auth.users(id) on delete set null, viewer_token text not null, created_at timestamptz not null default now(), constraint yofista_views_post_viewer_unique unique (post_id, viewer_token) ); create table if not exists public.yofista_follows ( follower_id uuid not null references auth.users(id) on delete cascade, following_id uuid not null references auth.users(id) on delete cascade, created_at timestamptz not null default now(), primary key (follower_id, following_id), constraint yofista_no_self_follow check (follower_id <> following_id) );
